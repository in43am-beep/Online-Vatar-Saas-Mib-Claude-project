Step 2 — Replace AI33 Voice with ElevenLabs

ElevenLabs has a standard REST API — same pattern as AI33. Add it as a new voice provider.

class ElevenLabsVoice:
    BASE = "https://api.elevenlabs.io/v1"

    def synthesize(self, api_key, text, voice_id, out_path):
        """ElevenLabs TTS — same interface as AI33."""
        resp = requests.post(
            f"{self.BASE}/text-to-speech/{voice_id}",
            headers={
                "xi-api-key": api_key,
                "Content-Type": "application/json"
            },
            json={
                "text": text,
                "model_id": "eleven_multilingual_v2",
                "voice_settings": {
                    "stability": 0.5,
                    "similarity_boost": 0.85,
                    "style": 0.4,
                    "use_speaker_boost": True
                }
            },
            timeout=120
        )
        if resp.status_code == 429:
            raise KeyExhausted("ElevenLabs rate limit")
        if resp.status_code in (401, 403):
            raise KeyRejected("ElevenLabs key rejected")
        resp.raise_for_status()
        Path(out_path).write_bytes(resp.content)
        return out_path

def _split_into_subscenes(narration, beat_duration_seconds):
    """Split a beat's narration into ~45-second sub-scenes."""
    words = narration.split()
    total_words = len(words)
    words_per_subscene = max(80, int(total_words * 45 / beat_duration_seconds))
    
    subscenes = []
    for i in range(0, total_words, words_per_subscene):
        chunk = " ".join(words[i:i + words_per_subscene])
        subscenes.append(chunk)
    return subscenes

Step 4 — Parallel Image Generation (Speed + Same Cost)
from concurrent.futures import ThreadPoolExecutor, as_completed

def generate_all_images_parallel(subscene_tasks, secrets, cfg):
    """Generate all 40 images simultaneously — same cost, 4× faster."""
    results = [None] * len(subscene_tasks)
    
    with ThreadPoolExecutor(max_workers=5) as ex:
        future_to_idx = {
            ex.submit(_generate_one, task, secrets, cfg): i
            for i, task in enumerate(subscene_tasks)
        }
        for future in as_completed(future_to_idx):
            i = future_to_idx[future]
            try:
                results[i] = future.result()
            except Exception as e:
                results[i] = _local_fallback(subscene_tasks[i])
    
    return results
